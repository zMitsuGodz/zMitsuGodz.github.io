function doPost(e) { var sheetDb = SpreadsheetApp.getActiveSpreadsheet(); var data = JSON.parse(e.postData.contents); if (data.tipoAccion === "VOTO") { var sheetVotos = sheetDb.getSheetByName("Votos"); if (!sheetVotos) { sheetVotos = sheetDb.insertSheet("Votos"); sheetVotos.appendRow(["Fecha", "Nombre", "Apellido", "Código", "p1", "p2", "p3", "p4", "p5", "p6", "p7", "p8", "p9", "p10"]); } sheetVotos.appendRow([new Date(), data.nombre, data.apellido, data.codigo, data.p1, data.p2, data.p3, data.p4, data.p5, data.p6, data.p7, data.p8, data.p9, data.p10]); return ContentService.createTextOutput("Voto Guardado").setMimeType(ContentService.MimeType.TEXT); } else if (data.tipoAccion === "CONFESION") { var sheetConfesiones = sheetDb.getSheetByName("Confesiones"); if (!sheetConfesiones) { sheetConfesiones = sheetDb.insertSheet("Confesiones"); sheetConfesiones.appendRow(["Fecha", "Texto", "Imagen"]); } sheetConfesiones.appendRow([new Date(), data.texto, data.imagen]); return ContentService.createTextOutput("Confesion Guardada").setMimeType(ContentService.MimeType.TEXT); } else { var sheet = sheetDb.getSheets()[0]; sheet.appendRow([new Date(), data.a, data.d, data.s, data.txt, data.t]); return ContentService.createTextOutput("OK").setMimeType(ContentService.MimeType.TEXT); } } function doGet(e) { var sheetDb = SpreadsheetApp.getActiveSpreadsheet(); var sheetMuro = sheetDb.getSheets()[0]; var dataMuro = sheetMuro.getDataRange().getValues(); var stats = {}; var reviews = []; for (var i = 1; i < dataMuro.length; i++) { var autor = dataMuro[i][1], delegado = dataMuro[i][2], estrellas = dataMuro[i][3], texto = dataMuro[i][4], tipo = dataMuro[i][5]; if (estrellas !== "" && estrellas > 0) { if (!stats[delegado]) stats[delegado] = { p: 0, t: 0 }; stats[delegado].t += 1; if (estrellas >= 3) stats[delegado].p += 1; } if (autor && texto) reviews.push({ a: autor, d: delegado, s: estrellas, txt: texto, t: tipo }); } var objStats = {}; for (var key in stats) objStats[key] = Math.round((stats[key].p / stats[key].t) * 100); var sheetVotos = sheetDb.getSheetByName("Votos"); var conteoVotos = { p1:{}, p2:{}, p3:{}, p4:{}, p5:{}, p6:{}, p7:{}, p8:{}, p9:{}, p10:{} }; if (sheetVotos) { var dataVotos = sheetVotos.getDataRange().getValues(); var votosPorCodigo = {}; for (var i = 1; i < dataVotos.length; i++) { var cod = dataVotos[i][3]; if (!votosPorCodigo[cod]) votosPorCodigo[cod] = { p1:"", p2:"", p3:"", p4:"", p5:"", p6:"", p7:"", p8:"", p9:"", p10:"" }; for (var j = 1; j <= 10; j++) { var val = String(dataVotos[i][j+3]).trim().toUpperCase(); if (val && !votosPorCodigo[cod]["p"+j]) { votosPorCodigo[cod]["p"+j] = val; } } } for (var cod in votosPorCodigo) { for (var j = 1; j <= 10; j++) { var val = votosPorCodigo[cod]["p"+j]; if (val) { if (!conteoVotos["p"+j][val]) conteoVotos["p"+j][val] = 0; conteoVotos["p"+j][val]++; } } } } var sheetConfesiones = sheetDb.getSheetByName("Confesiones"); var confesiones = []; if (sheetConfesiones) { var dataConf = sheetConfesiones.getDataRange().getValues(); for (var i = 1; i < dataConf.length; i++) { confesiones.push({ txt: dataConf[i][1], img: dataConf[i][2] }); } } var result = { aprobacion: objStats, comentarios: reviews, votos: conteoVotos, confesiones: confesiones }; return ContentService.createTextOutput(JSON.stringify(result)).setMimeType(ContentService.MimeType.JSON); }
